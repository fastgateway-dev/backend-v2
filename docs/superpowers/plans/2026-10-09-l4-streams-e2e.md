```go
//go:build e2e
package main

import ("io"; "net")

// serveTCP accepts connections and echoes each one until the peer closes.
func serveTCP(ln net.Listener) {
	for {
		c, err := ln.Accept(); if err != nil { return }
		go func(conn net.Conn) { defer conn.Close(); io.Copy(conn, conn) }(c)
	}
}

// serveUDP echoes every datagram back to its sender.
func serveUDP(pc net.PacketConn) {
	buf := make([]byte, 65535)
	for {
		n, addr, err := pc.ReadFrom(buf); if err != nil { return }
		pc.WriteTo(buf[:n], addr)
	}
}

func readFull(c net.Conn, buf []byte) (int, error) { return io.ReadFull(c, buf) }
```

```go
//go:build e2e
package harness

import ("strconv"; "strings"; "testing")

// L4Supported reports whether the running Envoy Gateway version is >= 1.8,
// the verified line for L4 BackendTrafficPolicy. Unknown ("") → false (skip),
// so an unlabeled environment never gives false confidence.
func L4Supported(egVersion string) bool {
	maj, min, ok := parseMajorMinor(egVersion)
	if !ok { return false }
	return maj > 1 || (maj == 1 && min >= 8)
}

func SkipIfL4Unsupported(t *testing.T, egVersion string) {
	t.Helper()
	if !L4Supported(egVersion) {
		t.Skipf("L4 streams require Envoy Gateway >= 1.8 (verified line); got %q — skipping", egVersion)
	}
}

func parseMajorMinor(v string) (int, int, bool) {
	v = strings.TrimPrefix(strings.TrimSpace(v), "v")
	if v == "" { return 0, 0, false }
	parts := strings.SplitN(v, ".", 3)
	if len(parts) < 2 { return 0, 0, false }
	maj, err1 := strconv.Atoi(parts[0]); min, err2 := strconv.Atoi(parts[1])
	if err1 != nil || err2 != nil { return 0, 0, false }
	return maj, min, true
}
```

```go
package harness

import ("context"; "fmt"; "github.com/google/uuid"; "k8s.io/apimachinery/pkg/runtime/schema"
	"github.com/fastgateway-dev/backend-v2/internal/models")

var (
	TCPRouteGVR = schema.GroupVersionResource{Group: "gateway.networking.k8s.io", Version: "v1alpha2", Resource: "tcproutes"}
	UDPRouteGVR = schema.GroupVersionResource{Group: "gateway.networking.k8s.io", Version: "v1alpha2", Resource: "udproutes"}
)

// StreamRouteGVR maps an L4 transport to its CRD GVR.
func StreamRouteGVR(protocol models.RouteProtocol) schema.GroupVersionResource {
	if protocol == models.RouteProtocolUDP { return UDPRouteGVR }
	return TCPRouteGVR
}

func (a *API) CreateStreamTemplate(ctx context.Context, projectID, name string) (uuid.UUID, error) {
	enable := true
	body := map[string]any{"name": name, "enableStream": &enable, "exposureType": "LoadBalancer", "tlsMode": "no_tls"}
	var dt models.DomainTemplate
	if _, err := a.Do(ctx, "POST", fmt.Sprintf("/projects/%s/domain-templates", projectID), body, &dt); err != nil {
		return uuid.Nil, err
	}
	if dt.Status != models.DomainTemplateStatusActive {
		return uuid.Nil, fmt.Errorf("stream template %s not active: %s (%s)", name, dt.Status, dt.StatusMessage)
	}
	return dt.ID, nil
}

func (a *API) CreateStream(ctx context.Context, projectID, name, namespace string, templateID uuid.UUID) (models.Stream, error) {
	body := map[string]any{"name": name, "namespace": namespace, "gatewayTemplateId": templateID}
	var s models.Stream
	_, err := a.Do(ctx, "POST", fmt.Sprintf("/projects/%s/streams", projectID), body, &s)
	return s, err
}

func (a *API) CreateStreamRoute(ctx context.Context, projectID, streamID string, body any) (Route, error) {
	var r Route
	_, err := a.Do(ctx, "POST", fmt.Sprintf("/projects/%s/streams/%s/routes", projectID, streamID), body, &r)
	return r, err
}

func (a *API) DeployStreamRoute(ctx context.Context, projectID, streamID, routeID string) error {
	_, err := a.Do(ctx, "POST", fmt.Sprintf("/projects/%s/streams/%s/routes/%s/deploy", projectID, streamID, routeID), nil, nil)
	return err
}

func (a *API) DeleteStream(ctx context.Context, projectID, streamID string) error {
	_, err := a.Do(ctx, "DELETE", fmt.Sprintf("/projects/%s/streams/%s", projectID, streamID), nil, nil)
	if se, ok := err.(*StatusError); ok && se.StatusCode == 404 { return nil }
	return err
}
```

```go
//go:build e2e
package harness

import ("context"; "fmt"; "net"; "time")

func DialTCP(ctx context.Context, addr string, payload []byte, timeout time.Duration) ([]byte, error) {
	d := net.Dialer{Timeout: timeout}
	c, err := d.DialContext(ctx, "tcp", addr); if err != nil { return nil, fmt.Errorf("dial tcp %s: %w", addr, err) }
	defer c.Close()
	c.SetDeadline(time.Now().Add(timeout))
	if _, err := c.Write(payload); err != nil { return nil, err }
	buf := make([]byte, len(payload))
	n := 0
	for n < len(payload) { m, err := c.Read(buf[n:]); if err != nil { return nil, err }; n += m }
	return buf, nil
}

// DialUDP retries until an echo arrives or the deadline passes (UDP is lossy).
func DialUDP(ctx context.Context, addr string, payload []byte, timeout time.Duration) ([]byte, error) {
	deadline := time.Now().Add(timeout)
	var lastErr error
	for time.Now().Before(deadline) {
		c, err := net.Dial("udp", addr); if err != nil { return nil, err }
		c.SetDeadline(time.Now().Add(500 * time.Millisecond))
		if _, err := c.Write(payload); err != nil { c.Close(); lastErr = err; continue }
		buf := make([]byte, len(payload)+64)
		n, err := c.Read(buf); c.Close()
		if err == nil { return buf[:n], nil }
		lastErr = err
	}
	return nil, fmt.Errorf("udp echo %s: no reply before deadline: %w", addr, lastErr)
}
```
